    <script>
        // Floating Heart Particles Generator
        function createHearts() {
            const container = document.getElementById('hearts-container');
            const heartIcons = ['💖', '🌸', '✨', '💜', '🎀', '👑'];
            
            for (let i = 0; i < 20; i++) {
                const heart = document.createElement('div');
                heart.className = 'heart-particle';
                heart.innerText = heartIcons[Math.floor(Math.random() * heartIcons.length)];
                heart.style.left = Math.random() * 100 + 'vw';
                heart.style.animationDuration = (Math.random() * 5 + 5) + 's';
                heart.style.animationDelay = (Math.random() * 5) + 's';
                heart.style.fontSize = (Math.random() * 15 + 15) + 'px';
                container.appendChild(heart);
            }
        }
        createHearts();

        // Music Player Toggle
        const musicBtn = document.getElementById('musicToggle');
        const bgMusic = document.getElementById('bgMusic');
        const musicIcon = document.getElementById('musicIcon');
        const musicText = document.getElementById('musicText');
        let isPlaying = false;

        musicBtn.addEventListener('click', () => {
            if (!isPlaying) {
                bgMusic.play().then(() => {
                    isPlaying = true;
                    musicText.innerText = "Music On 🎵";
                    musicIcon.classList.add('animate-spin');
                }).catch(err => {
                    console.log("Audio play blocked", err);
                });
            } else {
                bgMusic.pause();
                isPlaying = false;
                musicText.innerText = "Play Music ✨";
                musicIcon.classList.remove('animate-spin');
            }
        });

        // Toggle Interactive Envelope
        function toggleEnvelope() {
            const envelope = document.getElementById('envelope');
            envelope.classList.toggle('open');
            triggerConfetti();
        }

        // Confetti Trigger Function
        function triggerConfetti() {
            if (typeof confetti === 'function') {
                confetti({
                    particleCount: 80,
                    spread: 70,
                    origin: { y: 0.6 },
                    colors: ['#ff69b4', '#8a2be2', '#ffd1dc', '#d8bfd8', '#ffffff']
                });
            }
        }

        // Reveal Reason Card Compliments
        function revealReason(cardElement) {
            const reasonText = cardElement.querySelector('.reason-text');
            const reasonContent = cardElement.querySelector('.reason-content');

            if (reasonContent.classList.contains('hidden')) {
                reasonText.classList.add('hidden');
                reasonContent.classList.remove('hidden');
                cardElement.classList.add('border-pink-400', 'bg-pink-50/80');
                triggerConfetti();
            }
        }

        // Blow Candle Interactive Function
        let candleBlown = false;
        function blowCandle() {
            const flame = document.getElementById('candleFlame');
            const wishText = document.getElementById('wishText');

            if (!candleBlown) {
                flame.style.display = 'none';
                wishText.innerText = "🎉 Wish Granted! May all your dreams come true! ✨";
                candleBlown = true;
                
                // Huge Celebration Confetti
                confetti({
                    particleCount: 150,
                    spread: 100,
                    origin: { y: 0.5 },
                    colors: ['#ff69b4', '#8a2be2', '#ffb6c1', '#ffffff']
                });
            }
        }
    </script>
